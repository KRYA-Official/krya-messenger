import { initializeApp } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-app.js"; import { getFirestore, doc, getDoc, setDoc, updateDoc, addDoc, collection, serverTimestamp } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-firestore.js"; // 🔥 आपका Firebase config const firebaseConfig = { apiKey: "YOUR_KEY", authDomain: "YOUR_DOMAIN", projectId: "YOUR_ID" }; const app = initializeApp(firebaseConfig); const db = getFirestore(app); // 🔥 GLOBAL USER let currentUserId = null; let currentEmail = null; // ========================== // 🚀 INIT WALLET // ========================== window.initWallet = async function(userId, email){ currentUserId = userId; currentEmail = email; const ref = doc(db, "users", userId); const snap = await getDoc(ref); // अगर user नया है if(!snap.exists()){ await setDoc(ref, { email: email, coins: 50, winning: 0, createdAt: serverTimestamp() }); } loadWallet(); } // ========================== // 📥 LOAD WALLET // ========================== async function loadWallet(){ const snap = await getDoc(doc(db,"users",currentUserId)); const data = snap.data(); document.getElementById("coins").innerText = data.coins; document.getElementById("winning").innerText = data.winning; } // ========================== // ➕ ADD COINS // ========================== async function addCoins(amount, note="earn"){ const ref = doc(db,"users",currentUserId); const snap = await getDoc(ref); let newCoins = snap.data().coins + amount; await updateDoc(ref,{ coins:newCoins }); await addDoc(collection(db,"transactions"),{ userId: currentUserId, type:"earn", amount, note, time: serverTimestamp() }); loadWallet(); } // ========================== // ➖ SPEND COINS // ========================== async function spendCoins(amount, note="spend"){ const ref = doc(db,"users",currentUserId); const snap = await getDoc(ref); let coins = snap.data().coins; if(coins < amount){ alert("Coins कम हैं ❌"); return; } await updateDoc(ref,{ coins: coins - amount }); await addDoc(collection(db,"transactions"),{ userId: currentUserId, type:"spend", amount, note, time: serverTimestamp() }); loadWallet(); } // ========================== // 💰 ADD WINNING // ========================== async function addWinning(amount){ const ref = doc(db,"users",currentUserId); const snap = await getDoc(ref); let newWin = snap.data().winning + amount; await updateDoc(ref,{ winning:newWin }); await addDoc(collection(db,"transactions"),{ userId: currentUserId, type:"earn", amount, note:"winning", time: serverTimestamp() }); loadWallet(); } // ========================== // 🧪 TEST BUTTONS // ========================== window.testAddCoins = ()=> addCoins(10,"test reward"); window.testSpendCoins = ()=> spendCoins(5,"test spend"); window.testWin = ()=> addWinning(20);
